cmake_minimum_required(VERSION 3.22)
project(airpods-helper-kde LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.6 REQUIRED COMPONENTS Quick QuickControls2 Widgets DBus Test)
find_package(KF6Kirigami REQUIRED)
find_package(KF6StatusNotifierItem REQUIRED)
find_package(KF6DBusAddons REQUIRED)
find_package(LayerShellQt REQUIRED)
find_package(Plasma REQUIRED)
find_package(PlasmaQuick REQUIRED)
find_package(KF6ColorScheme REQUIRED)
find_package(KF6Config REQUIRED)
qt_standard_project_setup(REQUIRES 6.6)

qt_add_executable(airpods-helper-kde main.cpp client.cpp client.h battery.h panel.cpp panel.h icons.h)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../Cargo.toml" version_line REGEX "^version = ")
string(REGEX MATCH "[0-9]+\\.[0-9]+\\.[0-9]+[^\"]*" APP_VERSION "${version_line}")
target_compile_definitions(airpods-helper-kde PRIVATE APP_VERSION="${APP_VERSION}")
set(panel_qml qml/Main.qml qml/PanelContent.qml qml/Card.qml qml/StateSwitch.qml qml/DevicesPane.qml
    qml/SettingsPane.qml qml/Equalizer.qml qml/EqCurve.qml)
qt_add_qml_module(airpods-helper-kde URI AirPods QML_FILES ${panel_qml})
qt_add_resources(airpods-helper-kde "eq-math" PREFIX "/shared" BASE "../app/src" FILES ../app/src/eq.js)
set(panel_icons icons/airpods.svg icons/headphones.svg
    icons/off.svg icons/noise.svg icons/transparency.svg icons/adaptive.svg
    icons/charging.svg)
qt_add_resources(airpods-helper-kde "panel-icons" PREFIX "/" FILES ${panel_icons})
target_link_libraries(airpods-helper-kde PRIVATE Qt6::Quick Qt6::QuickControls2
    Qt6::Widgets Qt6::DBus KF6::StatusNotifierItem KF6::DBusAddons Plasma::PlasmaQuick LayerShellQt::Interface KF6::ColorScheme KF6::ConfigCore)
install(TARGETS airpods-helper-kde RUNTIME DESTINATION bin)
install(FILES org.costa.AirPodsHelper.desktop DESTINATION share/applications)
install(FILES ../app/src-tauri/icons/icon.svg
    DESTINATION share/icons/hicolor/scalable/apps RENAME org.costa.AirPodsHelper.svg)

# Load into the host Plasma process using its Qt and KDE libraries.
add_library(airpods-plasmoid MODULE plasmoid/applet.cpp client.cpp client.h)
target_include_directories(airpods-plasmoid PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(airpods-plasmoid PRIVATE Qt6::Quick Qt6::DBus Qt6::Gui
    Plasma::Plasma KF6::CoreAddons KF6::ColorScheme)
set_target_properties(airpods-plasmoid PROPERTIES PREFIX "" OUTPUT_NAME "org.costa.airpods"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/plasma/applets")
qt_add_resources(airpods-plasmoid "widget-ui" PREFIX "/qt/qml/AirPods" FILES ${panel_qml})
qt_add_resources(airpods-plasmoid "widget-icons" PREFIX "/" FILES ${panel_icons})
qt_add_resources(airpods-plasmoid "widget-eq" PREFIX "/shared" BASE "../app/src" FILES ../app/src/eq.js)
set(AIRPODS_PLASMA_PLUGIN_DIR "lib/qt6/plugins/plasma/applets" CACHE PATH "Plasma applet plugin install directory")
install(TARGETS airpods-plasmoid LIBRARY DESTINATION ${AIRPODS_PLASMA_PLUGIN_DIR} COMPONENT PlasmaWidget EXCLUDE_FROM_ALL)
install(DIRECTORY plasmoid/contents DESTINATION share/plasma/plasmoids/org.costa.airpods COMPONENT PlasmaWidget EXCLUDE_FROM_ALL)
install(FILES plasmoid/metadata.json DESTINATION share/plasma/plasmoids/org.costa.airpods COMPONENT PlasmaWidget EXCLUDE_FROM_ALL)

include(CTest)
if(BUILD_TESTING)
    configure_file(plasmoid/metadata.json "test-data/plasma/plasmoids/org.costa.airpods/metadata.json" COPYONLY)
    configure_file(plasmoid/contents/ui/main.qml "test-data/plasma/plasmoids/org.costa.airpods/contents/ui/main.qml" COPYONLY)
    qt_add_executable(plasmoid-test tests/plasmoid_test.cpp tests/mockbackend.h)
    target_compile_definitions(plasmoid-test PRIVATE WIDGET_TEST_DATA="${CMAKE_CURRENT_BINARY_DIR}/test-data"
        WIDGET_TEST_PLUGINS="${CMAKE_CURRENT_BINARY_DIR}")
    target_link_libraries(plasmoid-test PRIVATE Qt6::Test Qt6::Widgets Qt6::Quick Qt6::QuickControls2 Plasma::Plasma Plasma::PlasmaQuick KF6::Package KF6::ColorScheme)
    add_dependencies(plasmoid-test airpods-plasmoid)
    add_test(NAME plasma-widget COMMAND plasmoid-test)
    set_tests_properties(plasma-widget PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;DBUS_SESSION_BUS_ADDRESS=unix:path=/tmp/airpods-helper-no-bus;QT_ACCESSIBILITY=0")
    qt_add_executable(battery-test tests/battery_test.cpp battery.h)
    target_include_directories(battery-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(battery-test PRIVATE Qt6::Test KF6::ColorScheme)
    add_test(NAME battery COMMAND battery-test)
    qt_add_executable(panel-test tests/panel_test.cpp panel.cpp panel.h)
    target_include_directories(panel-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(panel-test PRIVATE Qt6::Test Qt6::Quick Qt6::Widgets Plasma::PlasmaQuick LayerShellQt::Interface)
    add_test(NAME panel-placement COMMAND panel-test)
    set_tests_properties(panel-placement PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;DBUS_SESSION_BUS_ADDRESS=unix:path=/tmp/airpods-helper-no-bus")
    qt_add_executable(qml-test tests/qml_test.cpp tests/mockbackend.h panel.cpp panel.h)
    target_include_directories(qml-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    qt_add_resources(qml-test "test-ui" PREFIX "/qt/qml/AirPods" FILES ${panel_qml})
    qt_add_resources(qml-test "test-eq" PREFIX "/shared" BASE "../app/src" FILES ../app/src/eq.js)
    qt_add_resources(qml-test "test-icons" PREFIX "/" FILES ${panel_icons})
    target_link_libraries(qml-test PRIVATE Qt6::Test Qt6::Widgets Qt6::Quick Qt6::QuickControls2 KF6::ColorScheme KF6::ConfigCore Plasma::PlasmaQuick LayerShellQt::Interface)
    foreach(scenario light dark disconnected disconnected-large headphones unknown large-font)
        add_test(NAME qml-${scenario} COMMAND qml-test windowLoads:${scenario})
        set_tests_properties(qml-${scenario} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;DBUS_SESSION_BUS_ADDRESS=unix:path=/tmp/airpods-helper-no-bus;QT_ACCESSIBILITY=0")
    endforeach()
    add_test(NAME qml-icons COMMAND qml-test iconsWorkWithoutGpuTinting)
    set_tests_properties(qml-icons PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;DBUS_SESSION_BUS_ADDRESS=unix:path=/tmp/airpods-helper-no-bus;QT_ACCESSIBILITY=0")
    qt_add_executable(client-test tests/client_test.cpp client.cpp client.h)
    target_include_directories(client-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(client-test PRIVATE Qt6::Test Qt6::DBus Qt6::Gui KF6::ColorScheme)
    find_program(DBUS_RUN_SESSION dbus-run-session REQUIRED)
    add_test(NAME dbus-client COMMAND ${DBUS_RUN_SESSION}
        --config-file=${CMAKE_CURRENT_SOURCE_DIR}/tests/session.conf -- $<TARGET_FILE:client-test>)
endif()
